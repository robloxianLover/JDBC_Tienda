Script para crear las tablas de la base de datos con SQL en Postgres:
<br>
-- This script was generated by the ERD tool in pgAdmin 4.
-- Please log an issue at https://github.com/pgadmin-org/pgadmin4/issues/new/choose if you find any bugs, including reproduction steps.
BEGIN;


CREATE TABLE IF NOT EXISTS tienda.categorias
(
    categoria_id bigserial NOT NULL,
    nombre character varying(80) COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT categorias_pkey PRIMARY KEY (categoria_id),
    CONSTRAINT unique_name UNIQUE (nombre)
);

CREATE TABLE IF NOT EXISTS tienda.clientes
(
    cliente_id bigserial NOT NULL,
    nombre character varying(80) COLLATE pg_catalog."default" NOT NULL,
    email character varying(120) COLLATE pg_catalog."default" NOT NULL,
    telefono character varying(30) COLLATE pg_catalog."default",
    creado_en timestamp with time zone DEFAULT now(),
    CONSTRAINT clientes_pkey PRIMARY KEY (cliente_id),
    CONSTRAINT unique_email UNIQUE (email)
);

CREATE TABLE IF NOT EXISTS tienda.detalles_pedido
(
    pedido_id bigint NOT NULL,
    producto_id bigint NOT NULL,
    cantidad integer NOT NULL,
    precio_unit numeric(10, 2) NOT NULL,
    CONSTRAINT detalles_pedido_pkey PRIMARY KEY (pedido_id, producto_id)
);

CREATE TABLE IF NOT EXISTS tienda.pagos
(
    pago_id bigserial NOT NULL,
    pedido_id bigint NOT NULL,
    monto numeric(10, 2) NOT NULL,
    metodo character varying(20) COLLATE pg_catalog."default" NOT NULL,
    status character varying(20) COLLATE pg_catalog."default" NOT NULL,
    fecha_pago timestamp with time zone DEFAULT now(),
    CONSTRAINT pagos_pkey PRIMARY KEY (pago_id)
);

CREATE TABLE IF NOT EXISTS tienda.pedidos
(
    pedido_id bigserial NOT NULL,
    cliente_id bigint NOT NULL,
    fecha timestamp with time zone DEFAULT now(),
    estado character varying(20) COLLATE pg_catalog."default" NOT NULL DEFAULT 'CREADO'::character varying,
    CONSTRAINT pedidos_pkey PRIMARY KEY (pedido_id)
);

CREATE TABLE IF NOT EXISTS tienda.productos
(
    product_id bigserial NOT NULL,
    categoria_id bigint,
    nombre character varying(120) COLLATE pg_catalog."default" NOT NULL,
    precio numeric(10, 2) NOT NULL,
    stock integer NOT NULL,
    sku character varying(40) COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT productos_pkey PRIMARY KEY (product_id)
);

ALTER TABLE IF EXISTS tienda.detalles_pedido
    ADD CONSTRAINT pedido_fk FOREIGN KEY (pedido_id)
    REFERENCES tienda.pedidos (pedido_id) MATCH SIMPLE
    ON UPDATE CASCADE
    ON DELETE CASCADE;


ALTER TABLE IF EXISTS tienda.detalles_pedido
    ADD CONSTRAINT products_fk FOREIGN KEY (producto_id)
    REFERENCES tienda.productos (product_id) MATCH SIMPLE
    ON UPDATE RESTRICT
    ON DELETE CASCADE;


ALTER TABLE IF EXISTS tienda.pagos
    ADD CONSTRAINT pedido_fk FOREIGN KEY (pedido_id)
    REFERENCES tienda.pedidos (pedido_id) MATCH SIMPLE
    ON UPDATE CASCADE
    ON DELETE CASCADE;


ALTER TABLE IF EXISTS tienda.pedidos
    ADD CONSTRAINT client_pedido_fk FOREIGN KEY (cliente_id)
    REFERENCES tienda.clientes (cliente_id) MATCH SIMPLE
    ON UPDATE CASCADE
    ON DELETE RESTRICT;


ALTER TABLE IF EXISTS tienda.productos
    ADD CONSTRAINT cat_prod_fk FOREIGN KEY (categoria_id)
    REFERENCES tienda.categorias (categoria_id) MATCH SIMPLE
    ON UPDATE CASCADE
    ON DELETE SET NULL;

END;
